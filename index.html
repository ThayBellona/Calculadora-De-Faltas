// O seu link exato do Boletim usando a sua matrícula const LINK_DO_SEU_BOLETIM = "https://suap.ifac.edu.br/edu/aluno/20241CRB12697480026/?tab=boletim"; document.getElementById('btnExtrair').addEventListener('click', () => { const divMensagem = document.getElementById('mensagem'); divMensagem.textContent = "Analisando..."; chrome.tabs.query({active: true, currentWindow: true}, (tabs) => { const abaAtual = tabs[0]; // 1. Verifica se você está na página exata do boletim if(!abaAtual.url.includes("tab=boletim")) { divMensagem.textContent = "Redirecionando para o seu Boletim..."; // Se não estiver, a extensão abre a página para você automaticamente! chrome.tabs.create({ url: LINK_DO_SEU_BOLETIM }); return; } // 2. Se já estiver no boletim, injeta o código para extrair os dados divMensagem.textContent = "Lendo dados do SUAP..."; chrome.scripting.executeScript({ target: {tabId: abaAtual.id}, function: extrairDadosDoSuap }, (resultados) => { if(resultados && resultados[0] && resultados[0].result) { const dados = resultados[0].result; if(dados.length === 0) { divMensagem.textContent = "Nenhuma falta encontrada. Tem certeza que o boletim carregou?"; return; } divMensagem.textContent = "Enviando para a Calculadora... 🚀"; // Transforma os dados e abre o SEU site oficial do GitHub const dadosCodificados = encodeURIComponent(JSON.stringify(dados)); const urlDestino = `https://thaybellona.github.io/Calculadora-De-Faltas/?importar=${dadosCodificados}`; chrome.tabs.create({ url: urlDestino }); } }); }); }); // A função espiã que roda dentro da página do SUAP function extrairDadosDoSuap() { const linhas = document.querySelectorAll('table[summary="Boletim do Aluno"] tbody tr'); let extraidas = []; linhas.forEach(tr => { const tdDisc = tr.querySelector('td[headers="th_disciplina"]'); const tdFalta = tr.querySelector('td[headers="th_total_faltas"]'); if(tdDisc && tdFalta) { let nomeSujo = tdDisc.textContent.trim(); // Pega o nome após o hífen let nomeLimpo = nomeSujo.split('-').pop().trim(); let qtdFaltas = parseInt(tdFalta.textContent.trim()) || 0; extraidas.push({ materia: nomeLimpo, faltas: qtdFaltas }); } }); return extraidas; }
